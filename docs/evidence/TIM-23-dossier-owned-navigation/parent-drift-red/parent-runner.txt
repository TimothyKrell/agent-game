import assert from 'node:assert/strict';
import { spawn } from 'node:child_process';
import { createHash } from 'node:crypto';
import { mkdir, readFile, writeFile } from 'node:fs/promises';
import { createServer } from '/home/timothykrell/Code/agent-games/node_modules/vite/dist/node/index.js';

const root = '/home/timothykrell/Code/agent-games';
const output = '/tmp/opencode/Dossier-lead-ending';
await mkdir(output);
const results = [];

async function run(name, args, env = {}) {
  const child = spawn(process.execPath, args, { cwd: root, env: { ...process.env, ...env } });
  let text = '';
  child.stdout.on('data', (chunk) => { text += chunk; });
  child.stderr.on('data', (chunk) => { text += chunk; });
  const code = await new Promise((resolve, reject) => { child.on('error', reject); child.on('close', resolve); });
  await writeFile(`${output}/${name}.log`, text);
  results.push({ name, code });
  await writeFile(`${output}/results.json`, JSON.stringify(results, null, 2));
  console.log(`${name}: exit ${code}\n${text.slice(-2500)}`);
  assert.equal(code, 0, name);
}

for (const name of ['route', 'production']) {
  const original = await readFile(`${root}/.dossier/${name}.mjs`, 'utf8');
  const adapted = original
    .replaceAll("from '@playwright/test'", `from '${root}/node_modules/@playwright/test/index.mjs'`)
    .replaceAll("from 'vite'", `from '${root}/node_modules/vite/dist/node/index.js'`)
    .replaceAll('/tmp/opencode/dossier-fixture-cache', `${output}/fixture-cache`)
    .replaceAll('/tmp/opencode/dossier-component-build', `${output}/component-build`)
    .replaceAll('6291', '6391').replaceAll('6292', '6392');
  await writeFile(`${output}/${name}.mjs`, adapted);
  await writeFile(`${output}/${name}.source.json`, JSON.stringify({
    sourceSha256: createHash('sha256').update(original).digest('hex'),
    adaptedSha256: createHash('sha256').update(adapted).digest('hex'),
    changes: 'Dependency import paths, isolated fixture/component cache and ports only. Evidence supplied through supported environment variables.',
  }, null, 2));
}

const server = await createServer({ root, cacheDir: `${output}/vite-cache`,
  server: { host: '127.0.0.1', port: 6391, strictPort: true },
  optimizeDeps: { entries: ['index.html', '.dossier/browser.html', '.dossier/focus.html', '.tim11/foundations.html'] },
});
await server.listen();
try {
  await run('ending-dev', ['node_modules/@playwright/test/cli.js', 'test', '--config', '.dossier/ending.config.ts'], {
    DOSSIER_ORIGIN: 'http://127.0.0.1:6391', DOSSIER_ENDING_EVIDENCE: `${output}/ending-dev`,
  });
  await run('route-dev', [`${output}/route.mjs`], {
    DOSSIER_ORIGIN: 'http://127.0.0.1:6391', DOSSIER_EVIDENCE_DIR: `${output}/route-dev`,
  });
} finally {
  await server.close();
}
await run('production', [`${output}/production.mjs`], {
  DOSSIER_ENDING_EVIDENCE: `${output}/ending-production`,
  DOSSIER_PRODUCTION_EVIDENCE_DIR: `${output}/production`,
  DOSSIER_EVIDENCE_DIR: `${output}/route-production`,
});
